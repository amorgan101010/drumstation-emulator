cmake_minimum_required(VERSION 3.20)
project(drumstation LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
add_library(adsp2181 STATIC vendor/adsp2181/adsp2181_core.c)
target_include_directories(adsp2181 PUBLIC vendor/adsp2181)
add_library(mn1880 STATIC vendor/mn1880/mn1880.cpp)
target_include_directories(mn1880 PUBLIC vendor/mn1880)
# The board alone, with no host I/O: what an embedding host (the studio engine) links.
add_library(drumstation_core INTERFACE)
target_include_directories(drumstation_core INTERFACE src)
target_link_libraries(drumstation_core INTERFACE adsp2181 mn1880)
add_executable(drumstation src/main.cpp)
find_library(ALSA_LIBRARY asound REQUIRED)
target_link_libraries(drumstation PRIVATE drumstation_core ${ALSA_LIBRARY})
enable_testing()

add_executable(adsp_core_test vendor/adsp2181/adsp2181_core_test.c)
target_link_libraries(adsp_core_test PRIVATE adsp2181)
add_test(NAME adsp_core COMMAND adsp_core_test)
add_test(NAME realtime_help COMMAND drumstation --realtime --help)
# Two boards in one process, with drumstation_core alone (no ALSA); skipped without ROMs.
add_executable(core_two_boards tests/core_two_boards.cpp)
target_link_libraries(core_two_boards PRIVATE drumstation_core)
add_test(NAME core_two_boards COMMAND core_two_boards)
set_tests_properties(core_two_boards PROPERTIES ENVIRONMENT "DRUMSTATION_ROOT=${CMAKE_SOURCE_DIR}" SKIP_RETURN_CODE 77)
add_executable(panel_transport_test tests/panel_transport_test.cpp)
target_link_libraries(panel_transport_test PRIVATE drumstation_core)
add_test(NAME panel_transport COMMAND panel_transport_test)
set_tests_properties(panel_transport PROPERTIES ENVIRONMENT "DRUMSTATION_ROOT=${CMAKE_SOURCE_DIR}" SKIP_RETURN_CODE 77)
add_executable(adsp-dasm vendor/adsp2181/dasm.cpp)
add_executable(mn-dasm vendor/mn1880/dasm.cpp)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME firmware COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_firmware.py)
    set_tests_properties(firmware PROPERTIES ENVIRONMENT "DRUMSTATION_BINARY=$<TARGET_FILE:drumstation>")
    add_test(NAME sequencer_model COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_sequencer.py)

endif()
